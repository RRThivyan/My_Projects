2. File Type Safety (ingestion.py)
python
def load_files_to_docs(paths: List[str]) -> List[Document]:
    all_docs: List[Document] = []
    for path in paths:
        lower = path.lower()  # Case-insensitive check
        if lower.endswith(".pdf"):
            all_docs.extend(pdf_to_docs(path))
        elif lower.endswith(".docx"):
            all_docs.extend(docx_to_docs(path))
        elif lower.endswith((".png", ".jpg", ".jpeg")):
            all_docs.extend(image_to_docs(path))
    return all_docs
Purpose: Ensures only supported file types are processed, preventing crashes from unsupported formats.

6. JSON Validation (evaluation.py)
python
def validate_output_schema(obj: Dict[str, Any]) -> List[str]:
    errors: List[str] = []
    
    if not isinstance(obj, dict):
        return ["Output is not a JSON object"]
    
    for key in ["query", "use_cases", "assumptions", "missing_info"]:
        if key not in obj:
            errors.append(f"Missing top-level field: {key}")
    
    use_cases = obj.get("use_cases", [])
    if not isinstance(use_cases, list):
        errors.append("use_cases is not a list")
    elif len(use_cases) == 0:
        errors.append("use_cases is empty")
Purpose: Ensures LLM outputs conform to expected schema, catching malformed or incomplete responses.

7. JSON Parsing with Error Handling (qwen_vl_extractor.py)
python
def extract_json_blocks(text: str) -> List[dict]:
    blocks = re.findall(r"\{.*?\}", text, flags=re.DOTALL)
    parsed_blocks = []
    for b in blocks:
        try:
            parsed_blocks.append(json.loads(b))  # Try to parse
        except json.JSONDecodeError:
            continue  # Skip invalid JSON
    return parsed_blocks
Purpose: Gracefully handles malformed JSON from vision model outputs without crashing.
